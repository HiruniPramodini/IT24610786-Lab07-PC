#include <mpi.h>
#include <stdio.h>
#include <stdlib.h>

#define N 1000000

int main(int argc, char **argv)
{
    MPI_Init(&argc, &argv);

    int rank, size;

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (N % size != 0)
    {
        if (rank == 0)
        {
            printf("Error: Number of processes must evenly divide %d.\n", N);
        }

        MPI_Finalize();
        return 1;
    }

    int chunk_size = N / size;

    /*
     * Only root allocates the complete array.
     */
    int *array = NULL;

    if (rank == 0)
    {
        array = (int *)malloc(N * sizeof(int));

        if (array == NULL)
        {
            printf("Error: Memory allocation failed on root.\n");
            MPI_Abort(MPI_COMM_WORLD, 1);
        }

        for (int i = 0; i < N; i++)
        {
            array[i] = i + 1;
        }

        printf("Root filled array with values 1 to %d\n", N);
    }

    /*
     * Local chunk.
     */
    int *local_chunk =
        (int *)malloc(chunk_size * sizeof(int));

    if (local_chunk == NULL)
    {
        printf("Rank %d: Memory allocation failed.\n", rank);
        MPI_Abort(MPI_COMM_WORLD, 1);
    }

    double start = MPI_Wtime();

    /*
     * Scatter array chunks.
     */
    MPI_Scatter(
        array,
        chunk_size,
        MPI_INT,
        local_chunk,
        chunk_size,
        MPI_INT,
        0,
        MPI_COMM_WORLD
    );

    /*
     * Calculate local sum.
     */
    long long local_sum = 0;

    for (int i = 0; i < chunk_size; i++)
    {
        local_sum += local_chunk[i];
    }

    /*
     * MPI_Scan calculates the prefix sum.
     *
     * Rank 0:
     *     local_sum_0
     *
     * Rank 1:
     *     local_sum_0 + local_sum_1
     *
     * Rank 2:
     *     local_sum_0 + local_sum_1 + local_sum_2
     *
     * etc.
     */
    long long prefix_sum = 0;

    MPI_Scan(
        &local_sum,
        &prefix_sum,
        1,
        MPI_LONG_LONG,
        MPI_SUM,
        MPI_COMM_WORLD
    );

    /*
     * Sum of all chunks before this rank.
     */
    long long sum_before_me =
        prefix_sum - local_sum;

    /*
     * Verification value.
     *
     * K = last element handled by this rank.
     */
    long long K =
        (long long)(rank + 1) * chunk_size;

    long long expected_prefix =
        K * (K + 1) / 2;

    printf(
        "Rank %d: local_sum = %lld, prefix_sum = %lld, "
        "sum_before_me = %lld, expected_prefix = %lld, %s\n",
        rank,
        local_sum,
        prefix_sum,
        sum_before_me,
        expected_prefix,
        prefix_sum == expected_prefix ? "PASS" : "FAIL"
    );

    /*
     * Root prints timing.
     */
    if (rank == 0)
    {
        double elapsed = MPI_Wtime() - start;

        printf(
            "\n[Scan] Rank 0 prefix_sum = %lld\n",
            prefix_sum
        );

        printf(
            "[Scan] Note: The LAST rank should have "
            "prefix_sum = 500000500000\n"
        );

        printf(
            "[Scan] Time = %.4f sec\n",
            elapsed
        );
    }

    /*
     * To explicitly verify the last rank's result.
     */
    if (rank == size - 1)
    {
        long long expected =
            (long long)N * (N + 1) / 2;

        printf(
            "[Scan] Last rank prefix_sum = %lld\n",
            prefix_sum
        );

        printf(
            "[Scan] Expected total      = %lld\n",
            expected
        );

        printf(
            "[Scan] Correct?            = %s\n",
            prefix_sum == expected ? "YES" : "NO"
        );
    }

    free(local_chunk);

    if (rank == 0)
    {
        free(array);
    }

    MPI_Finalize();

    return 0;
}
